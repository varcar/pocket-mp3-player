Pocket MP3 player

Play music from a microSD card, watch the track timer on a tiny OLED, and control playback with three buttons.

Parts on the bench

  • Seeed Studio XIAO ESP32-C3 and USB-C data cable
  • DFPlayer Mini (HW-247) and a FAT32 microSD card, up to 32 GB
  • 1.3-inch SH1106 OLED, 128 × 64, four-pin I²C
  • Female 3.5 mm stereo jack and wired headphones
  • Three momentary push buttons and one 2.5 kΩ resistor
  • Breadboard and jumper wires for the first test

Wire the player

Unplug USB while wiring. Start with the USB-powered connection shown at the bottom of the diagram. Follow the pin names printed on each board.

Pin connections in text
  • OLED: XIAO 3V3 → VCC, GND → GND, D4 → SDA, D5 → SCL.
  • DFPlayer: XIAO 5V → VCC for USB operation, GND → GND, D6 → 2.5 kΩ resistor → RX, TX on DFPlayer → D7 on XIAO.
  • Buttons: D1 for Back, D2 for Play/Pause, D3 for Next. Connect the other switched contact of each button to ground.
  • Jack: DAC_L → tip, DAC_R → ring, GND → sleeve. Leave SPK1 and SPK2 open. Use an inserted stereo AUX cable and continuity mode with power off to identify the three lugs.

Optional battery build. Use a protected 3.7 V single-cell LiPo on the underside BAT+ and BAT− pads. Check polarity before connecting. The XIAO 5V pin only supplies the player with USB attached. The diagram’s 3V3-to-DFPlayer option still needs a battery-only playback test. Use one DFPlayer supply connection at a time.

Load the music

  1. Check the card’s file system. An existing FAT32 card can keep its photos. Back up its contents first if formatting is necessary.
  2. Create a folder named mp3 directly on the card. Copy actual MP3 files into it as 0001.mp3, 0002.mp3, and so on.
  3. Edit titles[] in the sketch. Keep one title per file in numeric order. With a single song, leave just one title.
  4. Eject the card and insert it into the unpowered DFPlayer.

Upload the sketch

  1. In Arduino IDE, install esp32 by Espressif Systems from Boards Manager. Select XIAO_ESP32C3 and its USB port.
  2. Install U8g2 and DFRobotDFPlayerMini from Library Manager.
  3. Open the sketch, set the titles, and upload. This build uses volume 30 for the prototype headphones. Start lower for a different pair by editing player.volume(30).

Read the Arduino code
#include <Wire.h>
#include <U8g2lib.h>
#include <DFRobotDFPlayerMini.h>

U8G2_SH1106_128X64_NONAME_F_HW_I2C oled(U8G2_R0, U8X8_PIN_NONE);
HardwareSerial audioSerial(1);
DFRobotDFPlayerMini player;

// One title per file, in numeric order starting at /mp3/0001.mp3.
const char *titles[] = {"First song", "Second song", "Third song"};
const int trackCount = sizeof(titles) / sizeof(titles[0]);
int track = 0;
enum State { Ready, Playing, Paused, Fault };
State state = Ready;
const char *fault = "";
uint32_t started, elapsed, lastFrame;
struct Button {
  uint8_t pin;
  bool raw = HIGH, stable = HIGH;
  uint32_t changed = 0;
  bool pressed() {
    bool sample = digitalRead(pin);
    if (sample != raw) {
      raw = sample;
      changed = millis();
    }
    if (millis() - changed >= 25 && stable != raw) {
      stable = raw;
      return stable == LOW;
    }
    return false;
  }
};
Button previous{D1}, toggle{D2}, next{D3};

void playTrack(int index) {
  track = (index + trackCount) % trackCount;
  player.playMp3Folder(track + 1);
  started = millis();
  elapsed = 0;
  state = Playing;
}

void drawScreen() {
  oled.clearBuffer();
  oled.setFont(u8g2_font_6x10_tf);
  if (state == Fault) {
    oled.drawStr(0, 20, fault);
    oled.drawStr(0, 39, "Check card & wiring");
    oled.drawStr(0, 56, "Then press reset");
  } else {
    const char *label = state == Playing ? "NOW PLAYING" : state == Paused ? "PAUSED" : "PRESS PLAY";
    char counter[16];
    snprintf(counter, sizeof(counter), "%d/%d", track + 1, trackCount);
    oled.drawStr(0, 9, label);
    oled.drawStr(128 - oled.getStrWidth(counter), 9, counter);
    oled.drawHLine(0, 12, 128);
    uint32_t timeMs = elapsed + (state == Playing ? millis() - started : 0);
    int width = oled.getStrWidth(titles[track]);
    int offset = width > 128 ? (timeMs / 80) % (width + 18) : 0;
    oled.drawStr(-offset, 26, titles[track]);
    if (width > 128) oled.drawStr(width + 18 - offset, 26, titles[track]);
    for (int i = 0; i < 14; ++i) {
      int h = state == Playing ? 3 + int(14 * (0.5 + 0.5 * sin(timeMs / 180.0 + i * 1.7))) : 2;
      oled.drawBox(i * 5, 49 - h, 3, h);
    }
    char timer[16];
    uint32_t seconds = timeMs / 1000;
    snprintf(timer, sizeof(timer), "%lu:%02lu", (unsigned long)(seconds / 60), (unsigned long)(seconds % 60));
    oled.drawStr(128 - oled.getStrWidth(timer), 48, timer);
    oled.drawHLine(0, 52, 128);
    oled.drawStr(0, 63, "BACK");
    oled.drawStr(49, 63, state == Playing ? "PAUSE" : "PLAY");
    oled.drawStr(103, 63, "NEXT");
  }
  oled.sendBuffer();
}

void setup() {
#if ARDUINO_USB_CDC_ON_BOOT
  Serial.begin(115200);
#endif
  pinMode(D1, INPUT_PULLUP);
  pinMode(D2, INPUT_PULLUP);
  pinMode(D3, INPUT_PULLUP);
  Wire.begin(D4, D5);
  oled.begin();
  audioSerial.begin(9600, SERIAL_8N1, D7, D6);
  delay(5000);
  player.begin(audioSerial, false, false);
  player.volume(30);
  delay(200);
  drawScreen();
}

void loop() {
  if (player.available()) {
    uint8_t type = player.readType();
    int value = player.read();
    if (type == DFPlayerPlayFinished && state == Playing && millis() - started > 400)
      playTrack(track + 1);
    if (type == DFPlayerCardRemoved) {
      state = Fault;
      fault = "SD card removed";
    }
    if (state == Fault && (type == DFPlayerCardInserted || type == DFPlayerCardOnline)) {
      state = Ready;
      track = 0;
      elapsed = 0;
    }
#if ARDUINO_USB_CDC_ON_BOOT
    if (type == DFPlayerError) Serial.printf("DFPlayer error %d\n", value);
#endif
  }
  if (state != Fault) {
    if (previous.pressed()) playTrack(track - 1);
    if (next.pressed()) playTrack(track + 1);
    if (toggle.pressed()) {
      if (state == Ready) playTrack(track);
      else if (state == Playing) {
        elapsed += millis() - started;
        player.pause();
        state = Paused;
      } else {
        player.start();
        started = millis();
        state = Playing;
      }
    }
  }
  if (millis() - lastFrame >= 80) {
    lastFrame = millis();
    drawScreen();
  }
}

First playback

Give the HW-247 five seconds to start. Press the middle button when the OLED shows PRESS PLAY. D2 pauses and resumes, D1 goes back, and D3 skips forward. The timer pauses with the music. The moving bars are decorative.

If the timer runs but you hear silence

Try the first song and check its exact path: /mp3/0001.mp3. While powered, briefly touch DFPlayer IO2 to GND and release to start standalone playback. If BUSY drops near 0 V, check the headphone jack wiring and volume. If playback works only in standalone mode, check D6 → resistor → RX, TX → D7, and common ground.
